Integrations

Connect any app to GateKeeper via OIDC or reverse proxy forward auth.

Use these when configuring an app that supports OIDC / OAuth2

Discovery document{{.BaseURL}}/.well-known/openid-configuration

Reverse proxy auth

Protect any service - no app changes needed
1. Define the middleware
http:
  middlewares:
    gk-auth:
      forwardAuth:
        address: "{{.BaseURL}}/auth/verify"
        authResponseHeaders:
          - X-Auth-User
          - X-Auth-Email
          - X-Auth-Groups
2. Apply to a router
http:
  routers:
    my-app:
      rule: "Host(`app.example.com`)"
      middlewares:
        - gk-auth@file
      service: my-app
On success GateKeeper sets X-Auth-User, X-Auth-Email, and X-Auth-Groups (comma-separated group names) headers on the forwarded request. Using credential injection? Also add Authorization to authResponseHeaders. 
server {
    listen 443 ssl;
    server_name app.example.com;

    location / {
        auth_request     /gk-auth;
        auth_request_set $auth_user   $upstream_http_x_auth_user;
        auth_request_set $auth_email  $upstream_http_x_auth_email;
        auth_request_set $auth_groups $upstream_http_x_auth_groups;
        auth_request_set $auth_cred   $upstream_http_authorization;
        proxy_set_header X-Auth-User   $auth_user;
        proxy_set_header X-Auth-Email  $auth_email;
        proxy_set_header X-Auth-Groups $auth_groups;
        proxy_set_header Authorization $auth_cred;
        proxy_pass       http://localhost:3000;
    }

    location = /gk-auth {
        internal;
        proxy_pass              {{.BaseURL}}/auth/verify;
        proxy_pass_request_body off;
        proxy_set_header        Content-Length "";
    }
}
Requires ngx_http_auth_request_module - included in most Nginx builds. The Authorization lines are only needed when using credential injection. 
app.example.com {
    forward_auth {{.BaseURL}} {
        uri /auth/verify
        copy_headers X-Auth-User X-Auth-Email X-Auth-Groups Authorization
    }
    reverse_proxy localhost:3000
}
Requires Caddy v2.5+ with the forward_auth directive. Authorization is only forwarded when using credential injection.

Access policies

Restrict specific routes to named user groups
Add ?policy=name to the verify URL. GateKeeper returns 403 if the user is not in that policy. Create and manage policies in Policies. 
http:
  middlewares:
    gk-open:
      forwardAuth:
        address: "{{.BaseURL}}/auth/verify"
        authResponseHeaders: [X-Auth-User, X-Auth-Email, X-Auth-Groups]
    gk-staff:
      forwardAuth:
        address: "{{.BaseURL}}/auth/verify?policy=staff"
        authResponseHeaders: [X-Auth-User, X-Auth-Email, X-Auth-Groups]
    gk-admins:
      forwardAuth:
        address: "{{.BaseURL}}/auth/verify?policy=admins"
        authResponseHeaders: [X-Auth-User, X-Auth-Email, X-Auth-Groups]
location /admin {
    auth_request /gk-admins;
    proxy_pass   http://localhost:3000;
}

location / {
    auth_request /gk-staff;
    proxy_pass   http://localhost:3000;
}

location = /gk-admins {
    internal;
    proxy_pass              {{.BaseURL}}/auth/verify?policy=admins;
    proxy_pass_request_body off;
    proxy_set_header        Content-Length "";
}

location = /gk-staff {
    internal;
    proxy_pass              {{.BaseURL}}/auth/verify?policy=staff;
    proxy_pass_request_body off;
    proxy_set_header        Content-Length "";
}
app.example.com {
    handle /admin* {
        forward_auth {{.BaseURL}} {
            uri /auth/verify?policy=admins
            copy_headers X-Auth-User X-Auth-Email X-Auth-Groups
        }
        reverse_proxy localhost:3000
    }

    handle {
        forward_auth {{.BaseURL}} {
            uri /auth/verify?policy=staff
            copy_headers X-Auth-User X-Auth-Email X-Auth-Groups
        }
        reverse_proxy localhost:3000
    }
}
